cmake_minimum_required(VERSION 3.16)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" PREVIEW_VERSION LIMIT_COUNT 1)
project(raycon_slam_preview VERSION ${PREVIEW_VERSION} LANGUAGES CXX)
add_executable(raycon_slam_preview cpp/src/main.cpp)
target_compile_features(raycon_slam_preview PRIVATE cxx_std_17)
target_include_directories(raycon_slam_preview PRIVATE cpp/include)
target_compile_definitions(raycon_slam_preview PRIVATE DEMO_VERSION="${PROJECT_VERSION}")
if(MSVC)
  target_compile_options(raycon_slam_preview PRIVATE /W4)
else()
  target_compile_options(raycon_slam_preview PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  add_test(NAME public_cpp_checks COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_cpp.py" "$<TARGET_FILE:raycon_slam_preview>")
endif()
